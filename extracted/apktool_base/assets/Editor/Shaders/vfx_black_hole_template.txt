void vfx_fn_fragment(inout VFXMaterialInputs material, VFXEmitter emitter) {
    float radius = max(vfxParams.radius, 0.0001);
    float distanceToEmitter = distance(getUserWorldPosition().xyz, emitter.position);
    float innerRadius = radius * 0.2;
    float falloff = clamp((distanceToEmitter - innerRadius) / max(radius - innerRadius, 0.0001), 0.0, 1.0);

    float redFactor = pow(falloff, 0.80);
    float greenFactor = pow(falloff, 1.80);
    float blueFactor = pow(falloff, 3.20);

    material.baseColor.rgb *= float3(redFactor, greenFactor, blueFactor);
    material.ambientOcclusion *= falloff;
    material.reflectance *= falloff;
}

void vfx_fn_vertex(inout MaterialVertexInputs v, VFXEmitter emitter) {
    float radius = max(vfxParams.radius, 0.0001);
    float3 sourceWorldPosition = getUserWorldPosition();
    float3 toEmitter = emitter.position - sourceWorldPosition;
    float distanceToEmitter = length(toEmitter);
    float proximity = clamp(1.0 - (distanceToEmitter / radius), 0.0, 1.0);

    float suctionStrength = pow(proximity, 3.00);
    float orbitStrength = pow(proximity, 5.00);
    float3 directionToCenter = toEmitter / max(distanceToEmitter, 0.0001);
    float3 orbitAxis = (abs(directionToCenter.y) < 0.999) ? float3(0.0, 1.0, 0.0) : float3(1.0, 0.0, 0.0);
    float3 orbitDirectionRaw = cross(orbitAxis, directionToCenter);
    float orbitDirectionLength = length(orbitDirectionRaw);
    float3 orbitDirection = (orbitDirectionLength > 0.0001)
        ? orbitDirectionRaw / orbitDirectionLength
        : float3(1.0, 0.0, 0.0);

    float remainingDistance = distanceToEmitter;
    float suctionAmount = remainingDistance * 1.2 * suctionStrength;
    float orbitAmount = remainingDistance * 0.6 * orbitStrength;
    float3 displacement = (directionToCenter * suctionAmount) + (orbitDirection * orbitAmount);

    v.worldPosition.xyz += displacement;
}

Shader rendering engine is Google filament, search about Google filament shader attributes and uniforms

This is an example of a VFX Shader that distorts objects and changes their color, simulating a black hole.
This code darkens the pixel color on the surface and omits ambient light as it gets closer to the VFX emitter and pulls the vertices into the VFX emitter, becoming increasingly intense as the vertex approaches the emitter.
It runs on all rendered vertices and pixels of the scene.
Using emitter.position is merely a matter of choice.
A VFX effect can modify the entire scene without worrying about the emitter's position.

The dynamic time value is available as vfxParams.time and is not shown in the inspector.

Functions must begin with the prefix "vfx_fn_"
Structs must begin with the prefix "VFX_STRUCT_"

How to access parameters:
-Use vfxParams instead of materialParams.
-And it can only be accessed in the main function (vfx_fn_fragment or vfx_fn_vertex). If you need to access parameters in external functions, pass them as arguments.
