void vfx_fn_fragment(inout VFXMaterialInputs material, VFXEmitter emitter) {
    float lifeTime = max(vfxParams.lifeTime, 0.0001);
    float waveWidth = max(vfxParams.waveWidth, 0.0001);
    float speed = max(vfxParams.speed, 0.0001);

    float age = mod(vfxParams.time, lifeTime);
    float fade = 1.0 - clamp(age / lifeTime, 0.0, 1.0);
    float waveRadius = age * speed;
    float distanceToEmitter = distance(getUserWorldPosition().xyz, emitter.position);
    float wave = 1.0 - smoothstep(0.0, waveWidth, abs(distanceToEmitter - waveRadius));
    float intensity = wave * fade;

    float4 emissiveColor = float4(0.35 * intensity, 0.60 * intensity, 1.00 * intensity, 1.0);
    float emissiveNits = 10.0;
    float4 emissiveRadiance = toEmissiveRadiance(emissiveColor, emissiveNits);
    material.emissive = sumEmissiveRadiance(material.emissive, emissiveRadiance, emissiveNits);

    material.baseColor.rgb += float3(0.35, 0.60, 1.00) * intensity;
    material.ambientOcclusion = mix(material.ambientOcclusion, 1.0, intensity * 0.5);
}

void vfx_fn_vertex(inout MaterialVertexInputs v, VFXEmitter emitter) {
    float lifeTime = max(vfxParams.lifeTime, 0.0001);
    float waveWidth = max(vfxParams.waveWidth, 0.0001);
    float speed = max(vfxParams.speed, 0.0001);

    float age = mod(vfxParams.time, lifeTime);
    float fade = 1.0 - clamp(age / lifeTime, 0.0, 1.0);
    float waveRadius = age * speed;

    float3 sourceWorldPosition = getUserWorldPosition();
    float3 toEmitter = emitter.position - sourceWorldPosition;
    float distanceToEmitter = length(toEmitter);
    float3 directionToEmitter = toEmitter / max(distanceToEmitter, 0.0001);
    float wave = 1.0 - smoothstep(0.0, waveWidth, abs(distanceToEmitter - waveRadius));
    float pullStrength = wave * fade * waveWidth;

    v.worldPosition.xyz += directionToEmitter * pullStrength;
}
